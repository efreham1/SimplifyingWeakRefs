\begin{frame}{Garbage Collection (GC)}
    \begin{columns}[T]
        \column{0.55\textwidth}
        \begin{itemize}
            \item Java has \textbf{automatic memory management} $\to$ no \texttt{free()} or \texttt{delete}

\begin{frame}{Weak References}
    \begin{columns}[T]
        \column{0.52\textwidth}
        \small
        \begin{itemize}
            \item Strong references keep objects alive. If you can reach it, it won't be collected
            \item But sometimes you want to \textbf{observe} an object \emph{only while it is still alive}
            \item But if you hold a strong reference to the object, you keep it alive forever, defeating the purpose
        \end{itemize}
        \vspace{0.4em}
        \begin{alertblock}{The problem}
            \small
            You want to hold a reference that \textbf{does not} prevent garbage collection but still lets you read the object if it hasn't been collected yet.
        \end{alertblock}

        \column{0.48\textwidth}
        \centering
        \begin{tikzpicture}[scale=0.88]

\section{Motivation}
% =============================================================== %

% -------------------------------------------------------------------
% The problem
% -------------------------------------------------------------------
\begin{frame}{The Problem with Queue-less Weak References}
    \begin{columns}[T]
        \column{0.60\textwidth}
        \begin{itemize}
            \item GC performance directly impacts application performance
            \item ZGC was designed for high performance -- every unnecessary operation matters
        \end{itemize}
        \vspace{0.4em}
        \begin{alertblock}{The issue}
            \small
            When a \texttt{WeakReference} \emph{without} a \texttt{ReferenceQueue} is
            cleared, it is still handed to the \texttt{ReferenceHandler} thread --
            which then silently drops it, since there is no queue.\\[4pt]
            \textbf{Unnecessary work for every single queue-less weak reference.}
        \end{alertblock}
        \column{0.38\textwidth}
        \centering
        \begin{tikzpicture}[scale=1.0]
            \node[pipenode, minimum width=3.8cm] (all) at (1.9,5.0) {All WeakRefs};
            \node[pipenode, fill=yellow!15, minimum width=3.8cm] (proc) at (1.9,3.75) {Process};

            \node[draw=black!25, rounded corners=3pt, fill=green!8,

            \node[pipenode, fill=green!10, draw=alivegrn!50, minimum width=1.9cm]

        \column{0.50\textwidth}
        \small

        \only<1>{%
        \begin{itemize}
            \item Clearing a referent = writing \texttt{null} to the referent field
            \item The GC does this with a \textbf{CAS} -- an atomic instruction that only writes if the current value matches an expected value
            \item Needed because the ZGC runs concurrently with the application -- both may access the referent field at the same time without explicit synchronisation.
        \end{itemize}
        }%

        \only<2>{%
        \begin{itemize}
            \item \textbf{Key observation}: the only thing an app thread can write to a \texttt{WeakReference}'s referent field is \texttt{null} (via \texttt{WeakReference.clear()})
            \item If a race occurs, both sides write \texttt{null} -- the result is identical either way
            \item $\Rightarrow$ The CAS is unnecessary; a \textbf{plain store} is safe
        \end{itemize}
        }%
    \end{columns}

% -------------------------------------------------------------------
% Answering the Research Questions
% -------------------------------------------------------------------
\begin{frame}{Conclusions -- RQ1}
    \begin{block}{RQ1 -- Pipeline Optimisations}
        \small
        How do specific modifications to ZGC's processing of \texttt{WeakReference}
        objects \emph{without} a \texttt{ReferenceQueue} affect reference-processing
        time, total GC cycle time, GC memory footprint, and Java heap usage?
        \vspace{0.6em}
        \begin{itemize}
            \item \textbf{81\%} reduction in non-strong processing time (Clear Path + Dyn / All)
            \item Translates to only \textbf{8\%} in total major collection time
            \item Hard ceiling: non-strong is 14.3\% of major GC time (single) and 4.5\% (multi)
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Conclusions -- RQ2}
    \begin{block}{RQ2 -- Weak Fields}
        \small
        How does expressing weak semantics through annotated object fields affect
        reference processing time, GC cycle time, GC memory footprint, and Java heap
        usage compared to the optimised \texttt{WeakReference}-based approach of RQ1?
        \vspace{0.6em}
        \begin{itemize}
            \item \textbf{41\%} major GC time reduction (single), \textbf{28\%} (multi)
            \item \textbf{53\%} heap reduction (single) -- no \texttt{WeakReference} objects
            \item Structural advantage: fewer objects means less work across \emph{every} GC phase
        \end{itemize}
    \end{block}
\end{frame}

        \begin{alertblock}{Optimise the pipeline (All)}
            \small
            -81\% non-strong, but only -8\% total GC\\
            \textbf{Hard ceiling:} non-strong is just 14\% of total GC
        \end{alertblock}
        \vspace{0.4em}

            -41\% total GC (single)\\

        \end{alertblock}

            \small\centering
            Rethink how weak semantics is represented --\\
            not just how to process \texttt{WeakReference}s faster.
        \end{block}
