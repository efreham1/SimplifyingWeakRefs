\section{Introduction \& Motivation}
% ============================================================ %

\begin{frame}{GC Pauses Are a Real Cost}
    \begin{itemize}
        \item Java manages memory automatically: you allocate with \texttt{new},
              the garbage collector reclaims unreachable objects
        \item GC pauses disrupt latency-sensitive workloads --
              microservices, trading systems, real-time analytics
        \item \textbf{ZGC}: achieves sub-millisecond stop-the-world pauses
              regardless of heap size, by doing most GC work concurrently
    \end{itemize}
    \vspace{0.4em}
    \begin{alertblock}{One bottleneck remains}
        \small
        \textbf{Weak reference processing} cost scales linearly with the number of
        \texttt{WeakReference} objects. Large applications hold \emph{tens of millions} --
        yet ZGC applies the same costly pipeline to all of them, even when most
        do not need it.
    \end{alertblock}
\end{frame}

\begin{frame}{Why WeakReferences Are Everywhere}
    \begin{columns}
        \column{0.55\textwidth}
        \begin{itemize}
            \item A \texttt{WeakReference<T>} lets you hold a reference to an
                  object that the GC is \emph{still allowed} to collect
            \item Used in: \texttt{WeakHashMap}, Guava Cache, Caffeine,
                  many ORM frameworks, canonicalisation tables
            \item Optional \texttt{ReferenceQueue}: GC notifies you when
                  the referent is collected
            \item \textbf{Most} \texttt{WeakReference}s are created \emph{without}
                  a queue -- they only need weak reachability, not the callback
        \end{itemize}
        \column{0.45\textwidth}
        \begin{exampleblock}{Common (no queue)}
            \small\ttfamily
            new WeakReference<>(obj)\\[4pt]
            \small\rmfamily
            Pure weak reachability.\\No notification needed.
        \end{exampleblock}
        \vspace{0.4em}
        \begin{block}{Less common (with queue)}
            \small\ttfamily
            new WeakReference<>(obj, q)\\[4pt]
            \small\rmfamily
            GC notifies via \texttt{q} on collection.
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{Research Questions}
    \begin{block}{RQ1 -- Pipeline Optimisations}
        How do targeted modifications to ZGC's processing of \texttt{WeakReference}
        objects \emph{without} a \texttt{ReferenceQueue} affect reference-processing
        time, total GC cycle time, GC memory, and Java heap usage?
    \end{block}
    \vspace{0.8em}
    \begin{block}{RQ2 -- Weak Fields}
        How does expressing weak semantics through \emph{annotated object fields}
        affect the same metrics compared to the optimised \texttt{WeakReference}
        pipeline approach?
    \end{block}
\end{frame}

\section{Background}
% ============================================================ %

\begin{frame}{Java Memory Management and GC}
    \begin{columns}
        \column{0.58\textwidth}
        \begin{itemize}
            \item All Java objects live on the \textbf{heap}; you allocate with
                  \texttt{new} and the GC reclaims unreachable objects
            \item An object is \emph{live} if reachable from a \textbf{GC root}:
                  thread stacks, static fields, JNI references
            \item GC cycle: \textbf{mark} live objects $\to$
                  \textbf{process references} $\to$ \textbf{relocate}
            \item \textbf{Generational}: young gen (minor cycles, frequent)
                  + old gen (\textbf{major cycles}, infrequent)
        \end{itemize}
        \column{0.42\textwidth}
        \begin{block}{Object liveness}
            \small
            GC Root\\
            \quad $\blacktriangleright$ Object A \hfill \textcolor{codegreen}{\small live}\\
            \quad\quad $\blacktriangleright$ Object B \hfill \textcolor{codegreen}{\small live}\\[4pt]
            \textit{(unreachable)}\\
            \quad $\blacktriangleright$ Object C \hfill \textcolor{codered}{\small collected}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{Java Reference Types}
    \begin{columns}
        \column{0.52\textwidth}
        \begin{itemize}
            \item \textbf{Strong} (\texttt{Object o = new Foo()})\\
                  object lives as long as \texttt{o} is in scope
            \item \textbf{Weak} (\texttt{new WeakReference<>(obj)})\\
                  GC \emph{may} collect the referent;
                  \texttt{.get()} returns \texttt{null} afterwards
            \item \textbf{Soft}: like weak but GC uses a last-resort
                  heuristic (memory pressure)
            \item \textbf{Phantom}: always returns \texttt{null};
                  used for post-mortem resource cleanup
        \end{itemize}
        \column{0.48\textwidth}
        \begin{block}{ReferenceQueue}
            \small
            An optional notification channel: the GC deposits
            cleared \texttt{Reference} objects here.\\[4pt]
            Useful for explicit cleanup -- but the
            \emph{majority} of \texttt{WeakReference}s are
            queue-less.
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{ZGC: Concurrent, Sub-Millisecond GC}
    \begin{itemize}
        \item Most marking and relocation happens \textbf{concurrently} with
              the application -- stop-the-world pauses stay $<1$\,ms
        \item \textbf{Coloured pointers}: metadata bits embedded in the
              64-bit pointer itself (mark state, relocation status)
        \item \textbf{Load barriers}: small code sequences injected at pointer
              loads; let the GC update object state on-the-fly
        \item Handles heaps from 8\,GB to 16\,TB with the same pause budget
    \end{itemize}
    \begin{block}{Generational ZGC (JDK 21+)}
        \small
        Young generation collected by minor cycles;
        old generation by less frequent \textbf{major} cycles.
        Reference processing runs during the major-cycle marking pause.
    \end{block}
\end{frame}

\begin{frame}{Weak References in ZGC's Pipeline}
    \begin{enumerate}
        \item \textbf{Discovery} -- while marking, reference objects are appended
              to per-worker \emph{intrusive linked lists} via a hidden
              \texttt{discovered} field
        \item \textbf{Processing} -- check each referent's liveness; if unreachable,
              atomically clear the \texttt{referent} field (CAS + load barrier)
        \item \textbf{Enqueue} -- cleared references added to the \emph{pending list};
              the \texttt{ReferenceHandler} thread delivers them to their
              \texttt{ReferenceQueue}
    \end{enumerate}
    \vspace{0.4em}
    \begin{alertblock}{The cost of uniformity}
        \small
        Steps 1--3 apply to \emph{all} weak references -- including queue-less ones.
        A queue-less reference traverses the full pipeline only to be silently dropped
        at step~3, wasting linked-list traversal, CAS operations, and
        \texttt{ReferenceHandler} wake-ups.
    \end{alertblock}
\end{frame}

\section{Design \& Implementation}
% ============================================================ %

\begin{frame}{Four Mechanisms, Eight Variants}
    \begin{columns}
        \column{0.56\textwidth}
        \small
        Three \textbf{pipeline optimisations} targeting queue-less refs:
        \begin{itemize}
            \item \textbf{sep} -- split the discovered list at discovery time
            \item \textbf{dyn} -- replace the linked list with a dynamic array
            \item \textbf{clear\_path} -- specialised clear (no CAS)
        \end{itemize}
        \vspace{0.3em}
        One \textbf{structural alternative}:
        \begin{itemize}
            \item \textbf{weak\_fields} -- \texttt{@weak} field annotation
        \end{itemize}
        \vspace{0.3em}
        All $2^3 = 8$ pipeline combinations benchmarked independently.
        \column{0.44\textwidth}
        \small
        \begin{tabular}{lccc}
            \toprule
            \textbf{Variant} & \textbf{cp} & \textbf{sep} & \textbf{dyn} \\
            \midrule
            \texttt{none}     & -- & -- & -- \\
            \texttt{dyn}      & -- & -- & \checkmark \\
            \texttt{sep}      & -- & \checkmark & -- \\
            \texttt{sep\_dyn} & -- & \checkmark & \checkmark \\
            \texttt{cp}       & \checkmark & -- & -- \\
            \texttt{cp\_dyn}  & \checkmark & -- & \checkmark \\
            \texttt{cp\_sep}  & \checkmark & \checkmark & -- \\
            \texttt{all}      & \checkmark & \checkmark & \checkmark \\
            \bottomrule
        \end{tabular}
    \end{columns}
\end{frame}

\begin{frame}[fragile, shrink=5]{Mechanism 1: Skip-Enqueue Separation (sep)}
    \textbf{Idea:} At discovery time, route queue-less refs to a \emph{separate list}
    that never enters the pending list or \texttt{ReferenceHandler}.
    \begin{lstlisting}[language=C++, caption={Discovery split -- pseudocode}]
bool discover_reference(oop ref, ReferenceType type) {
    if (!should_discover(ref, type)) return false,
    if (type == REF_WEAK && !has_reference_queue(ref)) {
        // Queue-less: separate list, never enqueued
        no_queue_list_per_worker.append(ref),
    } else {
        // Queue-backed and all other types: standard path
        discovered_list_per_worker.append(ref),
    }
    mark_discovered(ref),  return true,
}
    \end{lstlisting}
    \begin{itemize}
        \item Per-worker lists -- no inter-thread contention during discovery
        \item Separate processing loop that clears but never enqueues
    \end{itemize}
\end{frame}

\begin{frame}[fragile, shrink=5]{Mechanism 2: Dynamic Array (dyn)}
    \textbf{Idea:} Replace the intrusive linked list with a contiguous array
    for cache-friendly traversal and pre-loaded referent data.
    \begin{lstlisting}[language=C++, caption={Array-based processing -- pseudocode}]
void process_no_queue_refs(ZWeakRefArray& arr) {
    size_t kept = 0,
    for (size_t i = 0, i < arr.length(), i++) {
        const ZWeakRefData& d = arr.at(i),  // sequential: cache-friendly
        d.mark_not_discovered(),
        if (try_make_inactive(d)) { cleared++, }
        else { kept++, }
    }
    arr.clear_and_reserve(kept),  // retain capacity for next cycle
}
    \end{lstlisting}
    \begin{itemize}
        \item Array grows on the C heap; amortised O(1) append during discovery
        \item Each entry \emph{pre-loads} the referent address and value
    \end{itemize}
\end{frame}

\begin{frame}{Mechanism 3: Specialised Clear Path (clear\_path)}
    \textbf{Idea:} For old-generation queue-less refs, replace the CAS-based
    clear with a plain store.
    \vspace{0.5em}
    \begin{columns}
        \column{0.48\textwidth}
        \begin{block}{Standard clear}
            \small
            1. Load barrier on referent\\
            2. CAS: referent $\to$ coloured null
        \end{block}
        \column{0.48\textwidth}
        \begin{block}{Specialised clear}
            \small
            1. Use pre-loaded value (no barrier)\\
            2. Direct store (no CAS)
        \end{block}
    \end{columns}
    \vspace{0.6em}
    \begin{alertblock}{Why safe?}
        During the relevant STW pause, no application thread can update the referent
        of an old-generation queue-less reference -- only the GC can.
        The CAS is therefore unnecessary.
    \end{alertblock}
\end{frame}

\begin{frame}{Superadditive Interaction: dyn + clear\_path}
    \begin{columns}
        \column{0.52\textwidth}
        \small
        \begin{itemize}
            \item \textbf{dyn alone}: $-$36\%
                  \quad{\tiny(cache locality, but still CAS)}
            \item \textbf{clear\_path alone}: $-$7\%
                  \quad{\tiny(faster clear, still linked list)}
            \item \textbf{dyn + clear\_path}: $\mathbf{-81\%}$
        \end{itemize}
        \vspace{0.4em}
        \begin{alertblock}{Why superadditive?}
            \small
            \texttt{dyn} pre-loads referent data during discovery.
            \texttt{clear\_path} \emph{uses} that data -- eliminating
            the load barrier \emph{and} the pointer-chase simultaneously.
        \end{alertblock}
        \column{0.48\textwidth}
        \includegraphics[width=\textwidth]{%
            composite_old_phase_concurrent_process_non-strong_slurm-5118807_combined_gc_single.pdf}
        \vspace{-0.5em}
        \scriptsize Non-strong processing time (single-object benchmark)
    \end{columns}
\end{frame}

\begin{frame}{Mechanism 4: Weak Fields (\texttt{@weak})}
    \textbf{Idea:} Weakness as a \emph{field property} -- no
    \texttt{WeakReference} wrapper object needed.
    \vspace{0.3em}
    \begin{block}{Usage}
        \small
        \texttt{import java.lang.ref.weak;} \quad
        \texttt{class Holder \{ @weak Target ref; \}}
    \end{block}
    \vspace{0.4em}
    \begin{itemize}
        \item Annotation recognised by the class-file parser (ZGC-only;
              changes across 27 files in compiler, interpreter, JIT, GC)
        \item Fields discovered during old-generation marking; stored in
              per-worker \texttt{ZWeakFieldArray}
        \item Unreachable referent: field zeroed directly --
              no pending list, no \texttt{ReferenceHandler}
        \item \textbf{Key:} no wrapper objects allocated or promoted to old gen
    \end{itemize}
\end{frame}

\section{Conclusions}
% ============================================================ %

\begin{frame}{Limitations}
    \begin{itemize}
        \item \textbf{Synthetic benchmarks}: single-object maximises ref-processing
              load; real applications have varied reference patterns and lifetimes
        \item \textbf{ZGC-specific}: all optimisations are ZGC-only; not applicable
              to G1GC or other collectors
        \item \textbf{One hardware platform}: UPPMAX supercomputer -- results
              may differ on desktop, server, or cloud hardware
        \item \textbf{\texttt{@weak} is a prototype}: changes span 27 files across
              compiler, interpreter, JIT, and GC -- not production-ready
    \end{itemize}
\end{frame}

\begin{frame}[shrink=5]{Conclusions}
    \begin{block}{RQ1 -- Pipeline Optimisations}
        \begin{itemize}
            \item \texttt{cp\_dyn} and \texttt{all}: \textbf{81\%} reduction in
                  non-strong processing time (superadditive interaction)
            \item Major collection improvement: at most \textbf{8\%} --
                  bounded by non-strong processing being only 14.3\% of total
        \end{itemize}
    \end{block}
    \vspace{0.3em}
    \begin{block}{RQ2 -- Weak Fields}
        \begin{itemize}
            \item \textbf{41\%} major GC time reduction; \textbf{53\%} heap reduction
            \item Structural advantage: eliminating wrapper objects benefits every
                  GC phase
        \end{itemize}
    \end{block}
    \vspace{0.3em}
    \begin{alertblock}{Overall Finding}
        Weak-reference overhead is a \textbf{representation problem},
        not a pipeline problem. Meaningful improvement requires rethinking
        how weak semantics is encoded in the language.
    \end{alertblock}
\end{frame}

\section{Background \& Motivation}
% ============================================================ %

\begin{frame}[shrink=3]{GC Performance in Managed Runtimes}
    \begin{itemize}
        \item GC pauses affect application latency; concurrent GC work competes
              with application threads for CPU and memory bandwidth
        \item \textbf{ZGC} achieves sub-millisecond pauses independent of heap size;
              marking, reference processing, and relocation are mostly concurrent
        \item One underoptimised component: \textbf{weak reference processing}\\
              -- cost scales linearly with the number of \texttt{WeakReference} objects
    \end{itemize}
    \begin{block}{Motivation}
        ZGC's reference-processing pipeline treats all weak references uniformly,
        even those that do not need the callback mechanism it provides.
    \end{block}
\end{frame}

\begin{frame}[shrink=5]{Weak References in Java}
    \begin{columns}
        \column{0.52\textwidth}
        \begin{itemize}
            \item \texttt{WeakReference<T>} allows objects to be reclaimed
                  despite being referenced
            \item Optional \texttt{ReferenceQueue}: callback when referent
                  is collected
            \item \textbf{Without a queue} (common case): pure weak
                  semantics, no GC callback needed
            \item Widely used in \texttt{WeakHashMap}, caches, observer patterns
        \end{itemize}

        \column{0.48\textwidth}
        \begin{block}{Without queue}
            \small\texttt{new WeakReference<>(obj)}\\[4pt]
            Only weak reachability required
        \end{block}
        \begin{exampleblock}{With queue}
            \small\texttt{new WeakReference<>(obj, q)}\\[4pt]
            GC notification when \texttt{obj} is cleared
        \end{exampleblock}
    \end{columns}
\end{frame}

\begin{frame}{ZGC's Reference-Processing Pipeline}
    \begin{enumerate}
        \item \textbf{Discovery} -- reference objects appended to per-worker
              \emph{intrusive linked lists} via hidden \texttt{discovered} field
        \item \textbf{Processing} -- check referent liveness; clear if unreachable
        \item \textbf{Enqueue} -- cleared references put on the \emph{pending list};
              \texttt{ReferenceHandler} delivers them to their \texttt{ReferenceQueue}
    \end{enumerate}
    \vspace{0.5em}
    \begin{alertblock}{The Problem}
        Steps 1--3 are applied \emph{uniformly} to all weak references --
        even queue-less ones that will never be enqueued. Wasted traversal
        and linked-list overhead for the most common case.
    \end{alertblock}
\end{frame}

\begin{frame}{Research Questions}
    \begin{block}{RQ1 -- Pipeline Optimisations}
        How do targeted modifications to ZGC's processing of \texttt{WeakReference}
        objects \emph{without} a \texttt{ReferenceQueue} affect reference-processing
        time, total GC cycle time, GC memory, and Java heap usage?
    \end{block}
    \vspace{0.8em}
    \begin{block}{RQ2 -- Weak Fields}
        How does expressing weak semantics through \emph{annotated object fields}
        affect the same metrics compared to the optimised \texttt{WeakReference}
        pipeline approach?
    \end{block}
\end{frame}

\begin{frame}{Four Mechanisms, Eight Variants}
    \begin{columns}

\begin{frame}[fragile, shrink=5]{Mechanism 1: Skip-Enqueue Separation}
    \textbf{Idea:} Route queue-less weak refs to a separate discovered list --
    bypasses the pending list and \texttt{ReferenceHandler} thread entirely.

    \begin{lstlisting}[language=C++, caption={Discovery split -- pseudocode}]
bool discover_reference(oop ref, ReferenceType type) {
    if (!should_discover(ref, type)) return false,

    if (type == REF_WEAK && !has_reference_queue(ref)) {
        // Queue-less: no-enqueue list, never touches ReferenceHandler
        no_queue_list_per_worker.append(ref),
    } else {
        // Queue-backed and all other types: standard pipeline
        discovered_list_per_worker.append(ref),
    }
    mark_discovered(ref),  return true,
}
    \end{lstlisting}
    \begin{itemize}
        \item Each GC worker has its own list -- no inter-thread contention
        \item Queue-less list processed by a specialised loop that never enqueues
    \end{itemize}
\end{frame}

\begin{frame}[fragile, shrink=5]{Mechanism 2: Dynamic Array}
    \textbf{Idea:} Replace the intrusive linked list with a contiguous
    \texttt{ZWeakRefArray} for cache-friendly traversal and pre-loaded referent data.

    \begin{lstlisting}[language=C++, caption={Array-based processing -- pseudocode}]
void process_no_queue_refs(ZWeakRefArray& arr) {
    size_t kept = 0,
    for (size_t i = 0, i < arr.length(), i++) {
        const ZWeakRefData& d = arr.at(i),  // sequential: cache-friendly
        d.mark_not_discovered(),
        if (try_make_inactive(d)) { cleared++, }
        else { kept++, }
    }
    arr.clear_and_reserve(kept),  // retain capacity for next cycle
}
    \end{lstlisting}
    \begin{itemize}
        \item Array grows on the C heap; amortised O(1) append per discovery
        \item Referent address and value \emph{pre-loaded} during discovery
    \end{itemize}
\end{frame}

\begin{frame}{Mechanism 3: Specialised Clear Path}
    \textbf{Idea:} For old-generation queue-less \texttt{WeakReference} objects,
    replace the CAS-based clear with a plain store.

    \vspace{0.5em}
    \begin{columns}
        \column{0.48\textwidth}
        \begin{block}{Standard clear path}
            \small
            1. Load barrier on referent\\
            2. CAS: referent $\to$ coloured null
        \end{block}

        \column{0.48\textwidth}
        \begin{block}{Specialised clear path}
            \small
            1. Use pre-loaded value (no barrier)\\
            2. Direct store (no CAS)
        \end{block}
    \end{columns}

    \vspace{0.6em}
    \begin{alertblock}{Why safe?}
        In the relevant STW pause, no application thread can update the referent
        of an old-generation queue-less reference -- only the GC can.
        The CAS is therefore unnecessary.
    \end{alertblock}
\end{frame}

\begin{frame}{Superadditive Interaction: dyn + clear\_path}
    \begin{columns}
        \column{0.52\textwidth}
        \small
        \begin{itemize}
            \item \textbf{dyn alone}: $-$36\%
                  \quad{\tiny(cache locality, but still CAS)}
            \item \textbf{clear\_path alone}: $-$7\%
                  \quad{\tiny(faster clear, still linked list)}
            \item \textbf{dyn + clear\_path}: $\mathbf{-81\%}$
        \end{itemize}
        \vspace{0.4em}
        \begin{alertblock}{Why superadditive?}
            \small
            \texttt{dyn} pre-loads referent data during discovery.\\[3pt]
            \texttt{clear\_path} uses pre-loaded data -- removing
            the load barrier \emph{and} the pointer-chase simultaneously.
        \end{alertblock}

        \column{0.48\textwidth}
        \includegraphics[width=\textwidth]{%
            composite_old_phase_concurrent_process_non-strong_slurm-5118807_combined_gc_single.pdf}
        \vspace{-0.5em}
        \scriptsize Non-strong time (single-object)
    \end{columns}
\end{frame}

\begin{frame}{Mechanism 4: Weak Fields}
    \textbf{Idea:} Weakness as a \emph{field property} -- no
    \texttt{WeakReference} wrapper object required.

    \vspace{0.3em}
    \begin{block}{Usage}
        \small
        \texttt{import java.lang.ref.weak;} \quad
        \texttt{class Holder \{ @weak Target ref; \}}
    \end{block}

    \vspace{0.4em}
    \begin{itemize}
        \item Annotation recognised by the class-file parser (ZGC-only;
              27 files across compiler, interpreter, JIT, GC)
        \item Fields discovered during old-generation marking; stored in
              per-worker \texttt{ZWeakFieldArray}
        \item Unreachable referent: annotated field zeroed directly --
              no pending list, no \texttt{ReferenceHandler}
        \item \textbf{Key:} no wrapper objects allocated or promoted
    \end{itemize}
\end{frame}

\section{Evaluation}

\begin{frame}{Benchmark Design}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Single-object}
        \begin{itemize}
            \item 20M holder objects pointing to one shared target
            \item Target released; \texttt{System.gc()} triggered
            \item Maximises per-cycle reference-processing load
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Multi-object}
        \begin{itemize}
            \item 2M objects, random-size payloads
            \item 5 rounds: 20\% of strong refs released per round
            \item More realistic liveness and locality patterns
        \end{itemize}
    \end{columns}

    \vspace{0.5em}
    \begin{block}{Execution Environment}
        \small
        UPPMAX supercomputer (SLURM), 48-core nodes, 100\,GB heap,
        \texttt{-XX:+UseZGC}, 250 outer iterations,
        4 instances pinned to isolated CPU sets.
    \end{block}
\end{frame}

\section{Discussion \& Conclusion}
% ============================================================ %

\begin{frame}[shrink=5]{Comparison with Other Languages}
    \small
    \centering
    \begin{tabular}{lll}
        \toprule
        \textbf{Language} & \textbf{Mechanism} & \textbf{Callback?} \\
        \midrule
        Java   & \texttt{WeakReference} + \texttt{ReferenceQueue} & Optional \\
        Python & \texttt{weakref.ref} + callable                   & Optional \\
        Go     & \texttt{weak.Pointer}                             & No \\
        C++    & \texttt{std::weak\_ptr}                           & No \\
        .NET   & \texttt{WeakReference<T>}                         & No \\
        Lua    & Weak tables (field-level)                         & No \\
        \bottomrule
    \end{tabular}

    \vspace{0.6em}
    \begin{itemize}
        \item Most languages treat weak reachability and lifecycle notification
              as separate concerns
        \item Java couples both into one object and one pipeline
        \item \texttt{@weak} brings Java closer to Lua's field-level model --
              weakness as a structural property of the field, not a wrapper
    \end{itemize}
\end{frame}

\begin{frame}[shrink=8]{Conclusions}
    \begin{block}{RQ1 -- Pipeline Optimisations}
        \begin{itemize}
            \item \texttt{cp\_dyn} and \texttt{all}: \textbf{81\%} reduction in
                  non-strong processing time (superadditive interaction)
            \item Major collection improvement: at most \textbf{8\%} --
                  bounded by the 14.3\% fraction non-strong processing occupies
        \end{itemize}
    \end{block}
    \vspace{0.3em}
    \begin{block}{RQ2 -- Weak Fields}
        \begin{itemize}
            \item \textbf{41\%} major GC time reduction;
                  \textbf{53\%} heap reduction
            \item Structural advantage: fewer objects in every GC phase
        \end{itemize}
    \end{block}
    \vspace{0.3em}
    \begin{alertblock}{Overall Finding}
        Weak-reference overhead is a \textbf{representation problem},
        not a pipeline problem. Meaningful improvement requires rethinking
        how weak semantics is encoded.
    \end{alertblock}
\end{frame}

\begin{frame}{Future Work}
    \begin{enumerate}
        \item \textbf{Production workloads} -- Java Flight Recorder on real
              applications with diverse reference patterns
        \item \textbf{ReferenceHandler load} -- latency impact for queue-backed
              references in mixed (queue-backed + queue-less) workloads
        \item \textbf{Extend clear\_path} -- to \texttt{SoftReference},
              \texttt{PhantomReference}, and queue-backed \texttt{WeakReference}
        \item \textbf{Upstream integration} -- submit via JDK-8029205
        \item \textbf{Harden \texttt{@weak}} -- test suite, formal correctness,
              upstream submission
    \end{enumerate}
\end{frame}
