        \begin{itemize}
            \item During marking: found \texttt{WeakReference}s go onto a per-worker linked list

            \item Cleared refs handed to the \texttt{ReferenceHandler} thread for queue delivery
            \item Queue-less refs travel through all three phases -- silently dropped at step~3
        \end{itemize}
    \end{columns}

\section{Motivation}
% =============================================================== %

% -------------------------------------------------------------------
% The problem
% -------------------------------------------------------------------
\begin{frame}{The Problem with Queue-less Weak References}
    \begin{columns}[T]
        \column{0.60\textwidth}
        \begin{itemize}
            \item GC performance directly impacts application performance
            \item ZGC was designed for low latency -- every unnecessary operation matters
            \item A JDK bug report (JDK-8029205) identified a concrete issue:
        \end{itemize}
        \vspace{0.4em}
        \begin{alertblock}{The issue}
            \small
            When a \texttt{WeakReference} \emph{without} a \texttt{ReferenceQueue} is
            cleared, it is still handed to the \texttt{ReferenceHandler} thread --
            which then silently drops it, since there is no queue.\\[4pt]
            \textbf{Unnecessary work for every single queue-less weak reference.}
        \end{alertblock}
        \column{0.38\textwidth}
        \centering
        \begin{tikzpicture}[scale=0.86]
            \node[pipenode, minimum width=3.8cm] (all) at (1.9,4.8) {All WeakRefs};
            \node[pipenode, fill=blue!8, minimum width=3.8cm] (proc) at (1.9,3.7) {Process};
            \node[pipenode, fill=yellow!15, minimum width=3.8cm] (enq) at (1.9,2.6) {Enqueue};
            \node[pipenode, fill=gray!20, minimum width=3.8cm] (rh) at (1.9,1.5) {ReferenceHandler};
            \node[pipenode, fill=red!12, minimum width=3.8cm, draw=codered] (gone) at (1.9,0.4) {Silently dropped};
            \draw[parrow] (all)--(proc);
            \draw[parrow] (proc)--(enq);
            \draw[parrow] (enq)--(rh);
            \draw[parrow, codered] (rh)--(gone);
            \node[font=\normalsize, codered, right=2pt] at (gone.east) {\texttimes};
            \node[font=\tiny\bfseries, codered, right=3pt] at (enq.east) {unnecessary};
        \end{tikzpicture}
    \end{columns}
\end{frame}

% -------------------------------------------------------------------
% Research Questions
% -------------------------------------------------------------------
\begin{frame}{Research Questions}
    \begin{block}{RQ1 -- Pipeline Optimisations}
        \small
        How do specific modifications to ZGC's processing of \texttt{WeakReference}
        objects \emph{without} a \texttt{ReferenceQueue} affect reference-processing
        time, total GC cycle time, GC memory footprint, and Java heap usage?
    \end{block}
    \vspace{0.6em}
    \begin{block}{RQ2 -- Weak Fields}
        \small
        How does expressing weak semantics through \emph{annotated object fields}
        affect the same metrics, compared to the optimised \texttt{WeakReference}
        pipeline approach?
    \end{block}
    \vspace{0.5em}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \small\textbf{RQ1:} optimise the \emph{existing} pipeline without changing
        how weak refs are represented
        \column{0.48\textwidth}
        \small\textbf{RQ2:} eliminate the \texttt{WeakReference} wrapper object
        entirely
    \end{columns}
\end{frame}

            \node[pipenode, fill=green!10, draw=alivegrn!60, minimum width=2.0cm]
                (st) at (3.9,2.2) {Plain store\\(null)};
            \draw[parrow, alivegrn!70] (pl)--(tk);
            \draw[parrow, alivegrn!70] (tk)--(st);

            \draw[<->,thick, dashed, black!35] (cas.east) -- (st.west);
            \node[font=\tiny, black!40] at (2.5,2.2) {same result};
        \end{tikzpicture}

        \column{0.46\textwidth}
        \small
        \begin{itemize}
            \item CAS guards against concurrent application writes
            \item \textbf{Key observation}: the only application-visible write to a \texttt{WeakReference}'s referent field is \texttt{WeakReference.clear()} -- which always writes \texttt{null}
            \item If a race occurs, both sides write \texttt{null} -- same outcome regardless of order
            \item $\Rightarrow$ The CAS is unnecessary; a plain store is safe
        \end{itemize}
        \vspace{0.3em}
        \begin{block}{\small Why only old-gen queue-less refs?}
            \scriptsize
            Only valid where the concurrent write is always \texttt{null}.
            A regular field can be written with any value.
        \end{block}

% -------------------------------------------------------------------
% Answering the Research Questions

\begin{frame}{Conclusions}
    \begin{block}{RQ1 -- Pipeline Optimisations}
        \begin{itemize}
            \item \textbf{81\%} reduction in non-strong processing time (\texttt{clear\_path\_dyn} / \texttt{all})
            \item Translates to only \textbf{8\%} in total major collection time
            \item Hard ceiling: non-strong is 14.3\% of major GC time (single) and 4.5\% (multi)
        \end{itemize}
    \end{block}
    \vspace{0.3em}
    \begin{block}{RQ2 -- Weak Fields}
        \begin{itemize}
            \item \textbf{41\%} major GC time reduction (single), \textbf{28\%} (multi)
            \item \textbf{53\%} heap reduction (single) -- no wrapper objects
            \item Structural advantage: fewer objects means less work across \emph{every} GC phase
        \end{itemize}
    \end{block}
\end{frame}

        \begin{block}{Optimise the pipeline (\texttt{all})}
            \small
            $-81$\% non-strong, but only $-8$\% total GC\\
            \textbf{Hard ceiling:} non-strong is only 14\% of major GC
        \end{block}
        \vspace{0.4em}
        \begin{block}{Change the representation (\texttt{weak\_fields})}
            \small
            $-41$\% total GC, $-53$\% Java heap\\
            Fewer objects $\to$ less work in \emph{every} GC phase
        \end{block}
        \vspace{0.4em}
        \begin{alertblock}{\centering \small The bottleneck is the \emph{representation}}
            \small\centering
            Rethink how weak semantics is represented --\\
            not just how to process the wrappers faster.
        \end{alertblock}
    \end{columns}
\end{frame}
